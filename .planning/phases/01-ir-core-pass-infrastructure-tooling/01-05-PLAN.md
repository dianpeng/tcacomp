    ```go
    package pkg

    // DefaultInstrumentations returns the standard compiler instrumentation
    // set — VerifyEach, IRDump (with real emitters wired), PassTiming — for a
    // caller-provided writer (Phase 5's cmd/ passes os.Stderr). Dump format is
    // routed by flags.F.DumpGraphviz; firing by flags.F.DumpAfter.
    func DefaultInstrumentations(w io.Writer) []pass.Instrumentation {
        dump := pass.NewIRDump(w) // constructor name per 01-02 SUMMARY
        dump.TextEmitter = ir.Print
        dump.DotEmitter = func(g *ir.Graph) string {
            var b strings.Builder
            _ = visualization.EmitDot(&b, g) // EmitDot never errors on a valid graph; documented emit-only
            return b.String()
        }
        return []pass.Instrumentation{
            pass.NewVerifyEach(w),
            dump,
            pass.NewPassTiming(w),
        }
    }
    ```
