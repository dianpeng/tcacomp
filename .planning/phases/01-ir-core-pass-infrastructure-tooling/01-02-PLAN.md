type Kind uint32

        KindSelect; KindCast; KindCall; KindEnd )

func (k Kind) Family() Family

type Node interface { BasePtr() *Base; KindOf() Kind }
type Base struct { /* id, kind, loc, lastTouchedBy, firstUse — no public mutators */ }

func (b *Base) LastTouchedBy() string
type Graph struct { /* name, params, block, nextID */ }
func (g *Graph) NewID() uint64
// Graph traversal for legality/verify walks: iterate g's single block's

| T-1-02-01 | Tampering | Unknown node kinds introduced by future passes slipping through verification | mitigate | Legality zero-value policy rejects any Kind absent from the window (including >= KindEnd) after EVERY pass; TestLegalityRejectsUnknownKind pins it (IR-03, Pitfall 8) |
