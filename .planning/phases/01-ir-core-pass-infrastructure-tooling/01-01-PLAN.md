// kind.go
type Kind uint32
const KindVersion = 1
// Phase-1 kinds (families marked): families grow in Phases 3-4 via new constants,
// never by renumbering (IR-03 forward-compat).
const (
    KindInvalid     Kind = iota
    KindIter             // abstract: induction variable @m (name, bound dim, init, step)
    KindIterCluster      // abstract: compound iter(@m, @k, @n) header
    KindMap              // abstract: map stage (operands, lambda, as-order)
    KindReduce           // abstract: reduce stage (init, piped input, combiner, annotation)
    KindStore            // abstract: store(target access, value)
    KindReturn           // abstract: kernel return
    KindConst            // scalar-expr: typed constant
    KindParam            // scalar-expr: lambda parameter reference
    KindBinOp            // scalar-expr: binary op (opcode table mirrors syntax.md §7.6)
    KindUnaryOp          // scalar-expr: unary op
    KindSelect           // scalar-expr: typed conditional ?:
    KindCast             // scalar-expr: explicit cast to named type
    KindCall             // scalar-expr: registered pure builtin call
    KindEnd              // sentinel: all kinds < KindEnd; PM rejects Kind >= KindEnd or KindInvalid (IR-03)
)
// Family marker interfaces (D-15, go/ast shape):
type Node interface { BasePtr() *Base; KindOf() Kind }
type ScalarExprNode interface { Node; scalarExprNode() }

// Opcode tables mirroring syntax.md §7.6 (used by NewBinOp/NewUnaryOp and the printer):
type BinOpCode int    // the §7.6 binary operator table
type UnaryOpCode int  // the §7.6 unary operator table
func (op BinOpCode) String() string
func (op UnaryOpCode) String() string

// node.go
type Loc struct { Span Span; PassOrigin string }  // Span is zero in Phase 1 (IR-06)
type Span struct { File string; Line, Col, EndLine, EndCol int }
type Base struct {
    id uint64; kind Kind; loc Loc
    lastTouchedBy string
    firstUse *Use   // head of this def's intrusive use-list (D-16)
}
func (b *Base) ID() uint64
func (b *Base) LastTouchedBy() string
// Param: a kernel I/O symbol (dim / tensor / scalar). Immutable after creation.
type Param struct { /* name + interned *Type + dim|tensor|scalar discriminant; unexported */ }
func (p Param) Name() string
func (p Param) Type() *Type
type Statement struct { id uint64 /* stable, D-04 */; def *SSADef; rhs Value }
type Block struct { statements []*Statement }   // exactly one per Graph (IR-02)
type Graph struct {
    name string
    params []Param   // dims, tensors, scalar params (kernel I/O symbols)
    block *Block
    nextID uint64    // monotonic; IDs never reused, stable across mutation (IR-05)
}
func (g *Graph) NewID() uint64
func (g *Graph) Name() string

// construct.go — the exported graph-construction surface. This is the ONLY way
// to create graphs and nodes from outside package ir (pkg/ builder, pass tests,
// visualization tests). Construction is exempt from MutContext (RESEARCH Pattern 2)
// but records PassOrigin so verifier diagnostics are meaningful from birth (D-17).
// Constructors deliberately do NOT validate kinds or operands: kind acceptance
// is the legality window's job (IR-03) and operand wiring is the verifier's job
// (PASS-06) — this lets cross-package tests construct out-of-window or unwired
// graphs to prove rejection (TestLegalityRejectsUnknownKind, TestBuildVerifiesThroughPM).
func NewGraph(name string) *Graph
func (g *Graph) AddDimParam(name string) Param
func (g *Graph) AddTensorParam(name string, t *Type) Param
func (g *Graph) AddScalarParam(name string, t *Type) Param
func (g *Graph) AddStatement(rhs Value) *Statement  // appends to the single block; ID from NewID (D-04)
// Low-level constructor: ANY kind (including >= KindEnd), operands wired as Uses.
func (g *Graph) NewNode(k Kind, origin string, operands ...Node) Node
// Typed wrappers (delegate to NewNode; the ergonomic surface for real producers):
func (g *Graph) NewConst(origin string, t *Type, v float64) Node
func (g *Graph) NewParamRef(origin string, p Param) Node
func (g *Graph) NewBinOp(origin string, op BinOpCode, a, b Node) Node
func (g *Graph) NewUnaryOp(origin string, op UnaryOpCode, x Node) Node
func (g *Graph) NewSelect(origin string, cond, a, b Node) Node
func (g *Graph) NewCast(origin string, t *Type, x Node) Node
func (g *Graph) NewCall(origin string, name string, args ...Node) Node
func (g *Graph) NewIterCluster(origin string, dims []Param, body Node) Node
func (g *Graph) NewIter(origin string, dim Param, bound Param, init, step Node) Node
func (g *Graph) NewMap(origin string, ops []Node, lam Lambda, as []Param) Node
func (g *Graph) NewReduce(origin string, init Node, piped Node, combiner Lambda) Node
func (g *Graph) NewStore(origin string, target Access, value Node) Node
func (g *Graph) NewReturn(origin string, value Node) Node
// Payload records for abstract stage nodes (immutable after construction):
type Lambda struct { Params []Param; Body Node }   // lambda(l, r): <body> — NOT a Kind
type Access struct { Tensor Param; Axes []Param }  // lhs[@m][@n]

// use.go — one Use per operand slot (IR-01/D-16). SSAUse = *Use;
// the def-producing node IS the SSA value; *Base is the shared SSAVar cell
// (design.md "SSADef and SSAUse shared an underlying pointer").
type Use struct {
    Val Node       // the def this operand points to
    User Node      // back-pointer to the owning node
    prev, next *Use
}
func UsesOf(def Node) []*Use   // snapshot slice (safe for splice-during-iteration)
func DefOf(u *Use) Node

// types.go — interned, pointer-compared (D-18): identity == equality, map-keyable.
type ElemType int
const ( TBool ElemType = iota; TF32; TF16; TBF16; TI8; TI16; TI32; TU8; TU16; TU32 )
// f8 intentionally absent (formats unspecified).
func (t ElemType) String() string  // "bool","f32","f16","bf16","i8","i16","i32","u8","u16","u32"
func InternScalar(t ElemType) *Type
func InternTensor(elem ElemType, dims []DimRef) *Type
// DimRef: extent 0 == dynamic/symbolic runtime dim; >0 == compile-time constant.

// mutate.go — the ONLY sanctioned mutation path (IR-04). Every function takes
// the pass context and stamps LastTouchedBy = ctx.Pass (D-17).
type MutContext struct{ Pass string }
func (ctx *MutContext) SetDef(u *Use, newVal Node) error
func (ctx *MutContext) AddUse(user Node, val Node) *Use
func (ctx *MutContext) RemoveUse(u *Use)
func (ctx *MutContext) ReplaceAllUsesWith(old, newVal Node) error  // self-RAUW no-op; O(uses) splice
func (ctx *MutContext) Erase(n Node) error  // refuses while uses remain; tombstones the ID

// verify.go — chain-consistency checkers (PASS-06 substrate).
type Violation struct { NodeID uint64; LastTouchedBy, Invariant, Msg string }
func Verify(g *Graph) []Violation   // deterministic order: BB statement order, then operand order
